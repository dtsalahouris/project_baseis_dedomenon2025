-- Έλεγχος κρατήσεων καταλυμάτων
SELECT ta.*, a.acc_NAME 
FROM tripAccommodation ta 
JOIN accommodation a ON ta.ta_ACC_ID = a.acc_ID;

-- Έλεγχος υπολογισμού κόστους από το trigger
SELECT ta_TR_ID, ta_NIGHTS, ta_ROOMS, ta_COST 
FROM tripAccommodation;

-- Έλεγχος συνολικού κόστους ανά ταξίδι
SELECT ta_TR_ID, SUM(ta_COST) AS total_accommodation_cost 
FROM tripAccommodation 
GROUP BY ta_TR_ID;


-- ============================================================
-- ΕΦΑΡΜΟΓΗ GUI (GRAPHICAL USER INTERFACE)
-- ============================================================
-- Η εφαρμογή GUI βρίσκεται στον φάκελο /gui και είναι υλοποιημένη
-- σε JavaFX. Πρόκειται για έναν γενικό database browser που επιτρέπει
-- την προβολή και διαχείριση όλων των πινάκων της βάσης δεδομένων.
--
-- ΔΟΜΗ ΑΡΧΕΙΩΝ:
-- gui/
-- ├── pom.xml                         -- Maven configuration (JavaFX 21, MySQL Connector)
-- └── src/main/java/app/
--     ├── MainApp.java                -- Entry point, δημιουργεί το παράθυρο 1200x700
--     ├── Db.java                     -- Σύνδεση MySQL (localhost:3306/travel_agency_2025)
--     └── TableBrowserView.java       -- Κύριο UI component (~450 γραμμές κώδικα)
--
-- ΤΕΧΝΙΚΕΣ ΛΕΠΤΟΜΕΡΕΙΕΣ:
-- • Framework: JavaFX 21.0.4 με Maven
-- • Database Driver: MySQL Connector/J 9.0.0
-- • Java Version: 17+
-- • UI Layout: BorderPane με SplitPane (70% TableView, 30% Form)
--
-- ΚΥΡΙΑ ΧΑΡΑΚΤΗΡΙΣΤΙΚΑ:
-- • Αυτόματη ανίχνευση Primary Keys, Foreign Keys και ENUM πεδίων
-- • Δυναμική δημιουργία φόρμας εισαγωγής/επεξεργασίας ανά πίνακα
-- • CRUD λειτουργίες (Insert, Update, Delete) για κάθε πίνακα
-- • Τα FK πεδία εμφανίζονται ως dropdowns με περιγραφικό κείμενο
--   π.χ. "1 (Athens City Tour)" αντί για απλό "1"
-- • Τα ENUM πεδία εμφανίζονται επίσης ως dropdowns
-- • Τα AUTO_INCREMENT πεδία είναι μόνο για ανάγνωση στην εισαγωγή
-- • Ειδικό φίλτρο χωρητικότητας για τον πίνακα vehicle
--
-- ΛΕΙΤΟΥΡΓΙΚΟΤΗΤΑ:
-- 1. Επιλογή πίνακα από dropdown → αυτόματη φόρτωση δεδομένων
-- 2. Click σε γραμμή → συμπλήρωση φόρμας με τα δεδομένα της
-- 3. Insert: Συμπλήρωση φόρμας και πάτημα "Insert"
-- 4. Update: Επιλογή γραμμής, τροποποίηση, πάτημα "Update"
-- 5. Delete: Επιλογή γραμμής και πάτημα "Delete"
--
-- ΕΚΤΕΛΕΣΗ:
-- cd gui && mvn clean javafx:run
--
-- ΣΗΜΕΙΩΣΗ: Η βάση δεδομένων πρέπει να τρέχει στο localhost:3306
-- με username "root" και κενό password.
