-- Βήμα 1: Πρώτα αναθέτουμε το vehicle 4 στο trip 15

-- Βήμα 2: Δημιουργούμε επικάλυψη - αλλάζουμε τις ημερομηνίες του trip 16
--         ώστε να επικαλύπτεται με το trip 15

SET tr_departure = '2025-10-11 07:00:00', 
    tr_return = '2025-10-13 21:00:00' 
WHERE tr_id = 16;

-- Βήμα 3: Προσπαθούμε να αναθέσουμε το ΙΔΙΟ vehicle (4) στο trip 16
-- Trip 15: 10-12 Οκτωβρίου 2025
-- Trip 16: 11-13 Οκτωβρίου 2025 (μετά την αλλαγή) → ΕΠΙΚΑΛΥΨΗ!

Αναμενόμενο αποτέλεσμα:
  - chk_no_overlap: FAIL
  - result_msg: NOT ASSIGNED ✗

-- Επαναφορά δεδομένων:
UPDATE trip 
SET tr_departure = '2025-10-25 07:30:00', 
    tr_return = '2025-10-29 21:30:00' 
WHERE tr_id = 16;
